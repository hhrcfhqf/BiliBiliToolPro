    21  function cleanupLegacyBBR() {
    22      const area = root();
    23      const walker = document.createTreeWalker(area, NodeFilter.SHOW_TEXT);
    24      const nodes = [];
    25      while (walker.nextNode()) nodes.push(walker.currentNode);
    26      for (const n of nodes) {
    27          if (!/\bBBR\b/i.test(n.nodeValue || "")) continue;
    28          const p = n.parentElement;

   160  async function save() {
   161      const el = ensure();
   162      const payload = {
   163          direction: el.querySelector("#ufw-direction")?.value,
   164          action: el.querySelector("#ufw-action")?.value,
   165          protocol: el.querySelector("#ufw-protocol")?.value,
   166          port: el.querySelector("#ufw-port")?.value,
   167          source: el.querySelector("#ufw-source")?.value || "Anywhere",
   168          comment: el.querySelector("#ufw-comment")?.value || ""
   169      };
   170      if (!payload.port) { alert("端口不能为空"); return; }
   171      const op = editing ? "修改" : "添加";
   172      if (!confirm(`确认${op} UFW 规则？\n${payload.action.toUpperCase()} ${payload.direction.toUpperCase()} ${payload.port}/${payload.protocol}`)) return;
   173
   174      try {
   175          if (!editing) {
   176              await api("/api/security/ufw/add", {method:"POST", body:JSON.stringify({...payload,confirm:"CONFIRM"})});
   177          } else {
   178              const n = editing.number;
   179              const ssh = await api("/api/security/ufw", {method:"GET"});
   180              const risky = (ssh.ssh_rule_numbers || []).includes(n) && String(payload.port) !== "22";
