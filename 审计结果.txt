    98  def _version_parts(v):
    99      s = str(v or "").strip().lower()
   100
   101      if s.startswith("v"):
   102          s = s[1:]
   103
   104      parts = re.findall(
   105          r"\d+|[a-z]+",
   106          s
   107      )
   108
   109      out = []
   110
   111      for p in parts:
   112          if p.isdigit():
   113              out.append((0, int(p)))
   114          else:
   115              out.append((1, p))
   116
   117      return out

   120  def _cmp(a, b):
   121      pa = _version_parts(a)
   122      pb = _version_parts(b)
   123
   124      n = max(len(pa), len(pb))
   125
   126      for i in range(n):
   127
   128          xa = pa[i] if i < len(pa) else (0, 0)
   129          xb = pb[i] if i < len(pb) else (0, 0)
   130
   131          if xa == xb:
   132              continue
   133
   134          if xa[0] != xb[0]:
   135              return -1 if xa[0] < xb[0] else 1
   136
   137          return -1 if xa[1] < xb[1] else 1
   138
   139      return 0
