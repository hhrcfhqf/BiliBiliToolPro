detect_os(){
  [[ -r /etc/os-release ]] || die "找不到 /etc/os-release"
  . /etc/os-release
  case "${ID:-}" in
    ubuntu|debian) ;;
    *) die "本版本只测试 Ubuntu/Debian。当前：${ID:-unknown}" ;;
  esac
  ARCH="$(dpkg --print-architecture 2>/dev/null || true)"
  [[ "$ARCH" == "arm64" || "$ARCH" == "amd64" ]] || die "仅支持 amd64/arm64；当前：$ARCH"
  info "OS=$ID $VERSION_ID ARCH=$ARCH"
}

apt_update_with_retry(){
  local attempts=0
  until apt-get update -y; do
    attempts=$((attempts+1))
    [[ $attempts -ge 5 ]] && die "apt-get update 连续失败 5 次"
    warn "apt-get update 失败，30 秒后重试（$attempts/5）"
    sleep 30
  done
}

apt_install_pkg(){
  local attempts=0
  until apt-get install -y --no-install-recommends "$@"; do
    attempts=$((attempts+1))
    [[ $attempts -ge 5 ]] && die "apt 安装失败：$*"
    warn "apt 安装失败，30 秒后重试（$attempts/5）"
    sleep 30
  done
}

apt_install(){
  export DEBIAN_FRONTEND=noninteractive
  apt_update_with_retry

    ca-certificates curl wget jq openssl git unzip tar gzip \
    python3 python3-venv ufw cron procps iproute2 dnsutils \
    lsof netcat-openbsd sed gawk coreutils
}

prompt(){
  local var="$1" text="$2" default="${3:-}" value
  if [[ -n "$default" ]]; then
    read -r -p "$text [$default]: " value || true
    value="${value:-$default}"
  else
    while [[ -z "${value:-}" ]]; do
      read -r -p "$text: " value || true
    done
  fi
  printf -v "$var" '%s' "$value"
}

validate_hostname(){
  [[ "$1" =~ ^[A-Za-z0-9.-]+$ ]] || die "域名格式不正确：$1"
}

cloudflare_api(){
  local method="$1" path="$2" data="${3:-}"
  if [[ -n "$data" ]]; then
    curl -fsS --retry 3 --connect-timeout 10 \
      -X "$method" \
      -H "Authorization: Bearer $CF_TOKEN" \
      -H "Content-Type: application/json" \
      "https://api.cloudflare.com/client/v4/$path" \
      --data "$data"
  else
    curl -fsS --retry 3 --connect-timeout 10 \
      -X "$method" \
      -H "Authorization: Bearer $CF_TOKEN" \
      -H "Content-Type: application/json" \
      "https://api.cloudflare.com/client/v4/$path"
  fi
}

cf_find_zone(){
  CF_ZONE_JSON="$(cloudflare_api GET "zones?name=$(python3 -c 'import urllib.parse,sys; print(urllib.parse.quote(sys.argv[1]))' "$ROOT_DOMAIN")&status=active&per_page=1")"
  CF_ZONE_ID="$(jq -r '.result[0].id // empty' <<<"$CF_ZONE_JSON")"
  [[ -n "$CF_ZONE_ID" ]] || die "Cloudflare 找不到 zone：$ROOT_DOMAIN。请确认域名已托管到 Cloudflare 且 Token 有 Zone 读取权限。"
}

cf_upsert_a(){
  local fqdn="$1" ip="$2" proxied="$3"
  local existing id payload result
  existing="$(cloudflare_api GET "zones/$CF_ZONE_ID/dns_records?type=A&name=$fqdn&per_page=100")"
  id="$(jq -r '.result[0].id // empty' <<<"$existing")"
  payload="$(jq -cn --arg type A --arg name "$fqdn" --arg content "$ip" --argjson proxied "$proxied" \
    '{type:$type,name:$name,content:$content,ttl:1,proxied:$proxied}')"
  if [[ -n "$id" ]]; then
    result="$(cloudflare_api PUT "zones/$CF_ZONE_ID/dns_records/$id" "$payload")"
  else
    result="$(cloudflare_api POST "zones/$CF_ZONE_ID/dns_records" "$payload")"
  fi
  [[ "$(jq -r '.success' <<<"$result")" == "true" ]] || die "Cloudflare DNS 更新失败：$fqdn"
  ok "Cloudflare DNS: $fqdn -> $ip proxied=$proxied"
}

setup_cloudflare(){
  echo
  echo "=== Cloudflare / 域名 ==="
  prompt ROOT_DOMAIN "你的 Cloudflare 根域名，例如 example.com"
  validate_hostname "$ROOT_DOMAIN"
  prompt NODE_HOST "节点域名" "node.$ROOT_DOMAIN"
  prompt PANEL_HOST "面板域名" "panel.$ROOT_DOMAIN"
  prompt REALITY_SNI "Reality handshake/SNI 目标（必须是你确认可用的公开 HTTPS 主机名）"
  validate_hostname "$NODE_HOST"
  validate_hostname "$PANEL_HOST"
  validate_hostname "$REALITY_SNI"

  echo
  echo "创建一个仅限当前 Zone 的 Cloudflare API Token。建议权限："
  echo "  Zone / DNS / Read"
  echo "  Zone / DNS / Edit"
  echo "Token 不要贴到聊天、脚本仓库或截图中。"
  prompt CF_TOKEN "Cloudflare API Token"
  [[ "$CF_TOKEN" =~ ^[A-Za-z0-9_-]+$ ]] || warn "Token 含有特殊字符；脚本仍会尝试使用。"

  cf_find_zone
  PUBLIC_IP="$(curl -4fsS --max-time 10 https://api.ipify.org || true)"
  [[ "$PUBLIC_IP" =~ ^[0-9.]+$ ]] || die "无法获取公网 IPv4。请确认 OCI 实例已联网。"

  cf_upsert_a "$NODE_HOST" "$PUBLIC_IP" false
  cf_upsert_a "$PANEL_HOST" "$PUBLIC_IP" true
}

install_acme(){
  if [[ ! -x /root/.acme.sh/acme.sh ]]; then
    info "安装 acme.sh"
    curl https://get.acme.sh | sh -s email="admin@$ROOT_DOMAIN"
  fi
  [[ -x /root/.acme.sh/acme.sh ]] || die "acme.sh 安装失败"
}

issue_cert(){
  info "通过 Cloudflare DNS API 申请 $NODE_HOST 证书"
  export CF_Token="$CF_TOKEN"
  /root/.acme.sh/acme.sh --set-default-ca --server letsencrypt >/dev/null
  /root/.acme.sh/acme.sh --issue --dns dns_cf \
    -d "$NODE_HOST" -d "$PANEL_HOST" --keylength ec-256
  mkdir -p "$CERT_DIR"
  /root/.acme.sh/acme.sh --install-cert -d "$NODE_HOST" --ecc \
    --fullchain-file "$CERT_DIR/fullchain.pem" \
    --key-file "$CERT_DIR/privkey.pem" \
    --reloadcmd "systemctl restart sing-box"
  chmod 600 "$CERT_DIR/privkey.pem"
  chmod 644 "$CERT_DIR/fullchain.pem"
  [[ -s "$CERT_DIR/fullchain.pem" ]] || die "证书链文件为空：$CERT_DIR/fullchain.pem"
  [[ -s "$CERT_DIR/privkey.pem" ]] || die "私钥文件为空：$CERT_DIR/privkey.pem"
  openssl x509 -in "$CERT_DIR/fullchain.pem" -noout -subject -issuer -dates >/dev/null || die "证书文件校验失败"
  ok "证书安装完成（包含节点域名和面板域名）"
}
