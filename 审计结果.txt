    44  def _validate_rule(d):
    45      if not isinstance(d, dict):
    46          raise ValueError("请求必须是 JSON object")
    47      direction = str(d.get("direction", "in")).strip().lower()
    48      action = str(d.get("action", "allow")).strip().lower()
    49      protocol = str(d.get("protocol", "tcp")).strip().lower()
    50      port = str(d.get("port", "")).strip()
    51      source = str(d.get("source", "Anywhere")).strip()
    52      comment = str(d.get("comment", "")).strip()
    53      if direction not in {"in", "out"}:
    54          raise ValueError("direction 必须为 in/out")
    55      if action not in {"allow", "deny", "reject", "limit"}:
    56          raise ValueError("action 非法")
    57      if protocol not in {"tcp", "udp", "both"}:
    58          raise ValueError("protocol 必须为 tcp/udp/both")
    59      import re as _re
    60      if not _re.fullmatch(r"\d{1,5}(?::\d{1,5})?", port):
    61          raise ValueError("port 非法")
    62      pp = [int(x) for x in port.split(":")]
    63      if not all(1 <= x <= 65535 for x in pp) or pp[0] > pp[-1]:
    64          raise ValueError("port 必须在 1-65535")
    65      if len(comment) > 80 or any(ord(x) < 32 or ord(x) == 127 for x in comment):
    66          raise ValueError("comment 非法")
    67      return {
    68          "direction": direction,
    69          "action": action,
    70          "protocol": protocol,
    71          "port": port,
    72          "source": source or "Anywhere",
    73          "comment": comment,

     5  export async function api(url, opt = {}) {
     6      try {
     7          opt = {
     8              credentials: "same-origin",
     9              ...opt,
    10              headers: {
    11                  ...(opt.headers || {})
    12              }
    13          };
    14
    15          if (opt.method === "POST") {
    16              opt.headers["Content-Type"] = "application/json";
    17
    18              let body = {};
    19
    20              if (opt.body) {
    21                  try {
    22                      body = JSON.parse(opt.body);
    23                  } catch (e) {
    24                      body = {};
    25                  }
    26              }
    27
    28              body.csrf = CSRF;
    29              opt.body = JSON.stringify(body);
    30          }
    31
    32          const r = await fetch(url, opt);
    33
    34          if (r.status === 401) {
    35              location = "/login";
    36              return null;
    37          }
    38
    39          const text = await r.text();
    40
    41          let data = null;
    42
    43          try {
    44              data = JSON.parse(text);
    45          } catch (e) {
    46              throw new Error(
    47                  "服务器返回的不是 JSON：HTTP " + r.status
    48              );
    49          }
    50
    51          if (!r.ok) {
    52              throw new Error(
    53                  data.error || ("HTTP " + r.status)
    54              );
    55          }
    56
    57          return data;
    58
    59      } catch (e) {
    60          console.error(url, e);
    61
    62          if (url !== "/api/session") {
    63              showError(e.message || String(e));
    64          }
    65
    66          return null;
    67      }
