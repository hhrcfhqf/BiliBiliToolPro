    39  function statusText(item) {
    40
    41      if (!item) {
    42          return "";
    43      }
    44
    45      if (item.status === "update_available") {
    46          return "有新版本";
    47      }
    48
    49      if (item.status === "up_to_date") {
    50          return "已是最新";
    51      }
    52
    53      if (item.status === "check_failed") {
    54          return "检查失败";
    55      }
    56
    57      return "";
    58  }

    61  function statusClass(item) {
    62
    63      if (!item) {
    64          return "";
    65      }
    66
    67      if (item.status === "update_available") {
    68          return "warn";
    69      }
    70
    71      if (item.status === "check_failed") {
    72          return "bad";
    73      }
    74
    75      if (item.status === "up_to_date") {
    76          return "ok";
    77      }
    78
    79      return "";
    80  }

   361  async function loadComponent(
   362      component,
   363      force = false
   364  ) {
   365
   366      let url =
   367          "/api/system/updates?component=" +
   368          encodeURIComponent(component);

   372      }
   373
   374      return await api(url);

   432  async function loadInitialComponents() {
   433
   434      const box =
   435          document.querySelector(
   436              "#oci-system-update"
   437          );
   438
   439      if (!box) {
   440          return;
   441      }
   442
   443      box.innerHTML = `
   444
   445          ${componentLoadingHtml(
   446              "📦 sing-box"
   447          )}
   448
   449          ${componentLoadingHtml(
   450              "🖥 Ubuntu / APT"
   451          )}
   452
   453          ${componentLoadingHtml(
   454              "🌐 Caddy"
   455          )}
   456
   457          ${componentLoadingHtml(
   458              "🤖 Ollama"
   459          )}
   460
   461      `;
   462
   463      /*
   464       * 四个模块独立读取缓存。
   465       *
   466       * 这里使用 Promise.all，
   467       * 但每个请求仍然是独立的 component。
   468       *
   469       * 不会调用总检查接口。
   470       */
   471
   472      const results =
   473          await Promise.allSettled([
   474              loadComponent("singbox", false),
   475              loadComponent("apt", false),
   476              loadComponent("caddy", false),
   477              loadComponent("ollama", false),
   478          ]);
   479

   481          return;
   482      }
   483
   484      const data = {
   485          singbox: null,
   486          apt: null,
   487          caddy: null,
   488          ollama: null,
   489      };
   490
   491      if (
   492          results[0].status === "fulfilled"
   493      ) {
   494          data.singbox =
   495              results[0].value;
   496      }
   497
   498      if (
   499          results[1].status === "fulfilled"
   500      ) {
   501          data.apt =
   502              results[1].value;
   503      }
   504
   505      if (
   506          results[2].status === "fulfilled"
   507      ) {
   508          data.caddy =
   509              results[2].value;
   510      }
   511
   512      if (
   513          results[3].status === "fulfilled"
   514      ) {
   515          data.ollama =
   516              results[3].value;
   517      }
   518
   519      renderData(data);
   520  }

   545  /* =========================================================
   546   * 单模块检查

   548
   549  async function checkSystemComponent(
   550      component
   551  ) {
   552
   553      if (checking[component]) {
   554          return;
   555      }
   556
   557      if (updating[component]) {
   558          return;
   559      }
   560
   561      const oldCard =
   562          document.querySelector(
   563              "#oci-update-card-" + component
   564          );
   565
   566      if (!oldCard) {
   567          return;
   568      }
   569
   570      checking[component] = true;
   571
   572      const button =
   573          oldCard.querySelector(
   574              "button"
   575          );
   576
   577      if (button) {
   578          button.disabled = true;
   579          button.textContent =
   580              "检查中……";
   581      }
   582
   583      try {
   584
   585          /*
   586           * force=true：
   587           * 只强制检查当前模块。
   588           */
   589
   590          const data =
   591              await loadComponent(
   592                  component,
   593                  true
   594              );
   595
   596          if (!data) {
   597              return;
   598          }
   599
   600          replaceComponent(
   601              component,
   602              data
   603          );
   604
   605      } catch (error) {
   606
   607          console.error(
   608              "system update check:",
   609              error
   610          );
   611
   612          showError(
   613              error.message ||
   614              String(error)
   615          );
   616
   617      } finally {
   618
   619          checking[component] = false;
   620      }
   621  }

   624  /* =========================================================
   625   * 单模块更新
   626   * ========================================================= */
   627
   628  async function updateSystemComponent(
   629      component
   630  ) {
   631
   632      if (checking[component]) {
   633          return;
   634      }
   635
   636      if (updating[component]) {
   637          return;
   638      }
   639
   640      const name =
   641          COMPONENT_NAMES[component] ||
   642          component;
   643
   644      if (
   645          !window.confirm(
   646              `确认更新 ${name}？\n\n` +
   647              `更新操作只针对当前模块，` +
   648              `不会更新其他模块。`
   649          )
   650      ) {
   651          return;
   652      }
   653
   654      updating[component] = true;
   655
   656      const card =
   657          document.querySelector(
   658              "#oci-update-card-" + component
   659          );
   660
   661      if (card) {
   662
   663          const buttons =
   664              card.querySelectorAll(
   665                  "button"
   666              );
   667
   668          buttons.forEach(
   669              button => {
   670                  button.disabled = true;
   671              }
   672          );
   673
   674          const updateButton =
   675              buttons.length >= 2
   676                  ? buttons[1]
   677                  : null;
   678
   679          if (updateButton) {
   680              updateButton.textContent =
   681                  "更新中……";
   682          }
   683      }
   684
   685      try {
   686
   687          /*
   688           * 新的独立更新接口。
   689           *
   690           * 不再使用旧的：
   691           * /api/action
   692           */
   693
   694          const result =
   695              await api(
   696                  "/api/system/update",
   697                  {
   698                      method: "POST",
   699
   700                      body: JSON.stringify({
   701                          component,
   702                          confirm: "CONFIRM",
   703                      }),
   704                  }
   705              );
   706
   707          if (!result) {
   708              return;
   709          }
   710
   711          if (!result.ok) {
   712
   713              throw new Error(
   714                  result.error ||
   715                  result.output ||
   716                  "更新失败"
   717              );
   718          }
   719
   720          /*
   721           * 更新完成后：
   722           *
   723           * 只重新检查当前模块。
   724           */
   725
   726          const data =
   727              await loadComponent(
   728                  component,
   729                  true
   730              );
   731
   732          if (data) {
   733
   734              replaceComponent(
   735                  component,
   736                  data
   737              );
   738          }
   739
   740          if (
   741              result.output &&
   742              result.output.includes(
   743                  "REBOOT_REQUIRED=1"
   744              )
   745          ) {
   746
   747              alert(
   748                  `${name} 更新完成。\n\n` +
   749                  `系统提示需要重启服务器。`
   750              );
   751
   752          } else {
   753
   754              alert(
   755                  `${name} 更新完成。`
   756              );
   757          }
   758
   759      } catch (error) {
   760
   761          console.error(
   762              "system update:",
   763              error
   764          );
   765
   766          showError(
   767              error.message ||
   768              String(error)
   769          );
   770
   771      } finally {
   772
   773          updating[component] = false;
   774      }
   775  }

   929  export function renderSystemUpdate() {
   930
   931      if (updateTimer) {
   932
   933          clearInterval(
   934              updateTimer
   935          );
   936
   937          updateTimer = null;
   938      }
   939
   940      /*
   941       * 首次进入：
   942       *
   943       * 四个模块分别读取自己的缓存。
   944       */
   945
   946      loadInitialComponents().catch(
   947          error => {
   948
   949              console.error(
   950                  "system update initial load:",
   951                  error
   952              );
   953
   954              showError(
   955                  error.message ||
   956                  String(error)
   957              );
   958          }
   959      );
   960
   961      /*
